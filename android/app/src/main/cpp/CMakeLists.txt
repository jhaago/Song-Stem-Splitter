cmake_minimum_required(VERSION 3.22)
project(vocal_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(vocal_pcm STATIC separator.cpp)
target_include_directories(vocal_pcm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
file(GLOB DEMUCS_SOURCES CONFIGURE_DEPENDS vendor/demucs/*.cpp)
add_library(demucs_inference STATIC ${DEMUCS_SOURCES})
target_include_directories(demucs_inference SYSTEM PUBLIC vendor/eigen vendor/demucs)
# IEEE finite checks are needed for input/model/output validation. Do not use -ffast-math.
target_compile_options(demucs_inference PRIVATE -O2 -Wno-unused-parameter)
target_compile_definitions(demucs_inference PUBLIC EIGEN_MPL2_ONLY EIGEN_DONT_PARALLELIZE)
add_library(vocal_backend STATIC demucs_backend.cpp)
target_link_libraries(vocal_backend PUBLIC vocal_pcm demucs_inference)
if(ANDROID)
 add_library(vocal_engine SHARED jni_bridge.cpp)
 target_link_libraries(vocal_engine PRIVATE vocal_backend log)
 # Native libraries use 16 KiB ELF alignment on modern Android devices.
 target_link_options(vocal_engine PRIVATE "-Wl,-z,max-page-size=16384")
else()
 enable_testing()
 add_executable(vocal_wrapper_tests ../../../../native-tests/wrapper_tests.cpp)
 target_link_libraries(vocal_wrapper_tests PRIVATE vocal_pcm)
 add_test(NAME wrapper COMMAND vocal_wrapper_tests)
 add_executable(vocal_inference ../../../../native-tests/inference.cpp)
 target_link_libraries(vocal_inference PRIVATE vocal_backend)
endif()
if(NOT ANDROID)
 add_executable(vocal_convolution_memory ../../../../native-tests/convolution_memory.cpp)
 target_link_libraries(vocal_convolution_memory PRIVATE demucs_inference)
 add_test(NAME decoder_memory COMMAND vocal_convolution_memory)
endif()
if(NOT ANDROID)
 add_executable(vocal_model_validation ../../../../native-tests/model_validation.cpp)
 target_link_libraries(vocal_model_validation PRIVATE vocal_backend)
 add_test(NAME model_validation COMMAND vocal_model_validation)
endif()
